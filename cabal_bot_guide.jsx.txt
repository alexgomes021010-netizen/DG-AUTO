function highlight(code) {
  return code
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/(#[^\n]*)/g, '<span style="color:#8b949e">$1</span>')
    .replace(
      /\b(import|from|def|class|return|if|elif|else|while|for|in|not|and|or|True|False|None|try|except|break|continue|pass|with|as|self|lambda|yield|async|await|raise|global|nonlocal)\b/g,
      '<span style="color:#ff7b72">$1</span>'
    )
    .replace(
      /("(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')/g,
      '<span style="color:#a5d6ff">$1</span>'
    )
    .replace(/\b(\d+\.?\d*)\b/g, '<span style="color:#f2cc60">$1</span>')
    .replace(
      /\b([A-Z][A-Za-z0-9_]*)\b/g,
      '<span style="color:#ffa657">$1</span>'
    );
}

import time

class BotState(Enum):
    IDLE        = auto()  # Parado, aguardando
    NAVIGATING  = auto()  # Andando até dungeon/mob
    COMBAT      = auto()  # Em combate
    COMBO       = auto()  # Executando sequência de habilidades
    LOOTING     = auto()  # Coletando drops
    HEALING     = auto()  # Usando poção/heal
    DUNGEON     = auto()  # Navegando dungeon automática
    DEAD        = auto()  # Morto, aguardando respawn

class BotCore:
    def __init__(self, memory, combo_engine, navigator, vision):
        self.state = BotState.IDLE
        self.memory = memory
        self.combo = combo_engine
        self.navigator = navigator
        self.vision = vision
        self.running = False
        self.target = None

    def tick(self):
        """Executado ~30x por segundo – coração do bot"""
        hp = self.memory.get_hp()
        mp = self.memory.get_mp()

        # Transições de estado
        if hp == 0:
            self.state = BotState.DEAD
        elif hp < 30 and self.state != BotState.HEALING:
            self.state = BotState.HEALING
        elif self.state == BotState.IDLE:
            self.target = self.vision.find_nearest_enemy()
            if self.target:
                self.state = BotState.NAVIGATING

        # Executa lógica do estado atual
        self._execute_state()

    def _execute_state(self):
        match self.state:
            case BotState.NAVIGATING:
                reached = self.navigator.move_to(self.target)
                if reached:
                    self.state = BotState.COMBAT
            case BotState.COMBAT:
                self.combo.auto_attack(self.target)
                enemy_dead = self.vision.is_enemy_dead()
                if enemy_dead:
                    self.state = BotState.LOOTING
            case BotState.COMBO:
                done = self.combo.execute_next_skill()
                if done:
                    self.state = BotState.COMBAT
            case BotState.HEALING:
                self.combo.use_potion()
                self.state = BotState.COMBAT
            case BotState.LOOTING:
                self.navigator.loot()
                self.state = BotState.IDLE
            case BotState.DEAD:
                time.sleep(5)  # Espera respawn
                self.state = BotState.IDLE

    def run(self):
        self.running = True
        print("[BOT] Iniciado!")
        while self.running:
            self.tick()
            time.sleep(1 / 30)  # 30 ticks/s`}

import pymem.process
import struct

class MemoryReader:
    def __init__(self, process_name: str):
        self.pm = None
        self.process_name = process_name
        self.base_addr = None

            "mp_max":      0x00A5BC1C,  # MP máximo
            "pos_x":       0x00A5BD00,  # Posição X no mapa

            "target_hp":   0x00A5C000,  # HP do alvo atual
            "in_combat":   0x00A5BE00,  # Flag: está em combate?

    def attach(self):
        """Conecta ao processo do jogo"""
        try:
            self.pm = pymem.Pymem(self.process_name)
            self.base_addr = pymem.process.module_from_name(

                self.process_name

            return True
        except Exception as e:
            print(f"[MEM] Falha: {e}")
            return False

    def _read_int(self, offset: int) -> int:
        """Lê um inteiro de 4 bytes na posição base + offset"""
        addr = self.base_addr + offset
        return self.pm.read_int(addr)

    def _read_float(self, offset: int) -> float:
        """Lê um float de 4 bytes"""
        addr = self.base_addr + offset
        return self.pm.read_float(addr)

        return self._read_int(self.OFFSETS["hp_current"])

    def get_hp_max(self) -> int:
        return self._read_int(self.OFFSETS["hp_max"])

    def get_hp_pct(self) -> float:
        """Retorna HP em percentual (0–100)"""
        max_hp = self.get_hp_max()
        return (self.get_hp() / max_hp * 100) if max_hp > 0 else 0

import time
import pyautogui
import keyboard
from dataclasses import dataclass, field
from typing import Callable

@dataclass
class Skill:
    name: str
    key: str               # Tecla do hotbar (ex: '1', '2', 'q', 'e')
    cooldown: float        # Segundos entre usos
    mp_cost: int           # Custo de MP
    priority: int          # Menor = maior prioridade
    condition: Callable    # Função que retorna True se pode usar
    last_used: float = field(default=0.0, init=False)

    def is_ready(self) -> bool:
        return time.time() - self.last_used >= self.cooldown

    def use(self):
        keyboard.press_and_release(self.key)
        self.last_used = time.time()
        print(f"  [SKILL] {self.name}")

class ComboEngine:
    def __init__(self, memory_reader):
        self.mem = memory_reader
        self.skills: list[Skill] = []
        self._setup_default_skills()

    def _setup_default_skills(self):
        """
        Monte seu combo aqui!
        Cada habilidade tem uma condição pra ser usada.
        """
        mp = lambda: self.mem.get_mp()
        hp_pct = lambda: self.mem.get_hp_pct()

        self.skills = [
            # ── Alta prioridade ──────────────────────────────
            Skill(
                name="Poção de HP",
                key='F1',
                cooldown=3.0,
                mp_cost=0,
                priority=0,
                condition=lambda: hp_pct() < 40,
            ),
            Skill(
                name="Buff de Ataque",
                key='b',
                cooldown=60.0,
                mp_cost=50,
                priority=1,
                condition=lambda: mp() > 50,
            ),
            # ── Habilidades de dano ──────────────────────────
            Skill(
                name="Golpe Fatal",    # Skill pesada, cooldown longo
                key='4',
                cooldown=10.0,
                mp_cost=80,
                priority=2,
                condition=lambda: mp() > 80,
            ),
            Skill(
                name="Corte Duplo",
                key='3',
                cooldown=4.0,
                mp_cost=40,
                priority=3,
                condition=lambda: mp() > 40,
            ),
            Skill(
                name="Slash Rápido",
                key='2',
                cooldown=2.0,
                mp_cost=20,

                key='1',
                cooldown=0.8,
                mp_cost=0,
                priority=5,
                condition=lambda: True,

    def execute_best_skill(self) -> bool:
        """Executa a melhor skill disponível (mais prioritária + pronta)"""

        candidates = sorted(
            [s for s in self.skills if s.is_ready() and s.condition()],
            key=lambda s: s.priority
        )
        if candidates:
            candidates[0].use()
            return True
        return False

        """
        Executa uma rotação fixa de habilidades em sequência.
        Útil para combos pré-definidos de classes do Cabal.
        """
        for key in rotation_keys:

import pyautogui
import keyboard

class DungeonNavigator:
    def __init__(self, memory, vision, combo):

        self.current_route = self.dungeon_routes.get(name, [])

        """Distância euclidiana entre posição atual e alvo"""
        cx, cy = self.mem.get_position()
        tx, ty = target

        Move o personagem em direção ao waypoint.
        Retorna True quando chegou.

        dy = math.sin(angle)

            timeout = time.time() + 30  # 30s máx por waypoint

            while not reached:
                if time.time() > timeout:
                    print("[DUNGEON] ⚠️  Timeout no waypoint, pulando...")
                    break

                # Verifica se há inimigos no caminho
                enemy = self.vision.find_nearest_enemy()
                if enemy:
                    print("[DUNGEON] ⚔️  Inimigo detectado – combatendo...")
                    self.combo.auto_attack_loop(duration=5.0)
                    self.loot()
                else:
                    reached = self.move_towards(wp)

                time.sleep(0.05)

            self.current_waypoint_idx += 1

        print("[DUNGEON] ✅ Dungeon concluída!")

        """Coleta drops próximos pressionando a tecla de loot"""
        print("[LOOT] Coletando drops...")

import cv2
import numpy as np
import pyautogui
from PIL import ImageGrab

class VisionEngine:
    def __init__(self):
        # Cor dos nomes de inimigo sobre eles (ajuste pra seu jogo)
        # Em jogos estilo Cabal, nomes de mob são geralmente vermelhos
        self.ENEMY_COLOR_LOWER = np.array([0,   100, 100])   # HSV min
        self.ENEMY_COLOR_UPPER = np.array([10,  255, 255])   # HSV max

        # Cor da barra de HP do inimigo (geralmente vermelho)
        self.HP_BAR_LOWER = np.array([0,   150, 150])
        self.HP_BAR_UPPER = np.array([8,   255, 255])

    def screenshot(self):
        """Captura a tela atual como array NumPy (BGR)"""
        img = ImageGrab.grab()
        return cv2.cvtColor(np.array(img), cv2.COLOR_RGB2BGR)

    def find_on_screen(self, template_path: str, threshold=0.8):
        """
        Template matching: encontra uma imagem na tela.
        Útil para botões, portais, ícones de UI.
        """
        screen = self.screenshot()
        template = cv2.imread(template_path, cv2.IMREAD_UNCHANGED)
        if template is None:
            return None

        # Converte para grayscale para matching mais rápido
        s_gray = cv2.cvtColor(screen, cv2.COLOR_BGR2GRAY)
        t_gray = cv2.cvtColor(template, cv2.COLOR_BGR2GRAY)

        result = cv2.matchTemplate(s_gray, t_gray, cv2.TM_CCOEFF_NORMED)
        _, max_val, _, max_loc = cv2.minMaxLoc(result)
